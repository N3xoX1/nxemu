cmake_minimum_required(VERSION 3.22)
if(POLICY CMP0141)
    cmake_policy(SET CMP0141 NEW)
endif()
project(NxemuIPOProbe LANGUAGES C CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../nxemu_android_ipo.cmake")
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NXEMU_PROBE_LINKER_OPTION)
    add_link_options("${NXEMU_PROBE_LINKER_OPTION}")
endif()
if(NXEMU_PROBE_CACHE_OPTION)
    add_link_options("${NXEMU_PROBE_CACHE_OPTION}")
endif()
add_library(ipo_c STATIC value.c)
add_library(ipo_cxx STATIC value.cpp)
add_library(ipo_shared SHARED shared.cpp)
target_link_libraries(ipo_shared PRIVATE ipo_cxx ipo_c)
set_target_properties(ipo_shared PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(ipo_probe main.cpp)
target_link_libraries(ipo_probe PRIVATE ipo_shared)
if(APPLE)
    enable_language(OBJCXX)
    target_sources(ipo_cxx PRIVATE value.mm)
    if(CMAKE_CXX_COMPILE_OPTIONS_IPO)
        target_compile_options(ipo_cxx PRIVATE
            "$<$<COMPILE_LANGUAGE:OBJCXX>:${CMAKE_CXX_COMPILE_OPTIONS_IPO}>")
    endif()
    target_compile_definitions(ipo_shared PRIVATE NXEMU_PROBE_OBJCXX=1)
endif()
